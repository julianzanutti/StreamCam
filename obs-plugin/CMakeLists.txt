cmake_minimum_required(VERSION 3.25)
project(StreamCamObs VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(OBS_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.tooling/obs-sdk" CACHE PATH "OBS 33.0.1 source headers")
set(OBS_IMPORT_LIBRARY "" CACHE FILEPATH "Import library generated from installed obs.dll")
if(NOT EXISTS "${OBS_IMPORT_LIBRARY}")
    message(FATAL_ERROR "Set OBS_IMPORT_LIBRARY to the import library for installed OBS")
endif()
add_library(streamcam-source MODULE streamcam-source.cpp)
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE_ROOT}/libobs/obsconfig.h.in" "${CMAKE_CURRENT_BINARY_DIR}/generated/obsconfig.h")
target_include_directories(streamcam-source PRIVATE "${OBS_SOURCE_ROOT}/libobs" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(streamcam-source PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(streamcam-source PRIVATE "${OBS_IMPORT_LIBRARY}")
set_target_properties(streamcam-source PROPERTIES PREFIX "")
add_executable(StreamCamObsFrameProbe frame-probe.cpp)
target_include_directories(StreamCamObsFrameProbe PRIVATE "${OBS_SOURCE_ROOT}/libobs" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(StreamCamObsFrameProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(StreamCamObsFrameProbe PRIVATE "${OBS_IMPORT_LIBRARY}")
